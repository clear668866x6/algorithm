
struct Targin {
    static constexpr int N = 2e4 + 10;
    int dfn[N], low[N], id[N];
    int stk[N], scc_cnt, scc_size[N];
    bool in_stk[N];
    int tmp, top;
    V<int> g[N];

    void add(int a, int b) {
        tmp = top = 0;
        g[a].eb(b);
    }

    void targin(int u) {
        dfn[u] = low[u] = ++tmp;
        stk[++top] = u, in_stk[u] = 1;

        for (auto j : g[u]) {
            if (!dfn[j]) {
                targin(j);
                low[u] = min(low[u], low[j]);
            } else if (in_stk[j]) {
                low[u] = min(low[u], dfn[j]);
            }
        }

        if (low[u] == dfn[u]) {
            int y;
            scc_cnt++;
            do {
                y = stk[top--];
                in_stk[y] = 0;
                id[y] = scc_cnt;
                scc_size[scc_cnt]++;
            } while (y != u);
        }
    }

} A;
