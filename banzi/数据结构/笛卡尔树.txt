笛卡尔树：
- 它能将序列的问题<=>树上问题。（比如abc435f）
- 注意：我们的笛卡尔树里面存的是序列的下标。
 
 
 
 V<int> l(n + 1, 0), r(n + 1, 0), stk;
    auto build = [&]() {
        int rt = 0;
        FOR(i, 1, n) {
            int lst = 0;
            while (!stk.empty() && w[stk.back()] > w[i]) {
                lst = stk.back();
                stk.pop_back();
            }
            if (!stk.empty())
                r[stk.back()] = i;
            else
                rt = i;
            l[i] = lst;
            stk.push_back(i);
        }
    };
