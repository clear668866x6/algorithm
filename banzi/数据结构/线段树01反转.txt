struct SegmentTree {
    static constexpr int N = 5e5 + 10;

    struct E {
        int l, r, c0, c1, tag;
    } tr[N << 2];

    void pushup(int u) {
        tr[u].c0 = tr[u << 1].c0 + tr[u << 1 | 1].c0;
        tr[u].c1 = tr[u << 1].c1 + tr[u << 1 | 1].c1;
    }

    void evel(int u, int tag) {
        swap(tr[u].c0, tr[u].c1);
        tr[u].tag ^= tag;
    }

    void pushdown(int u) {
        if (tr[u].tag) {
            evel(u << 1, tr[u].tag);
            evel(u << 1 | 1, tr[u].tag);
            tr[u].tag = 0;
        }
    }

    void build(int u, int l, int r, string &s) {
        tr[u] = {l, r};
        if (l == r) {
            if (s[l] == '1') {
                tr[u] = {l, r, 0, 1, 0};
            } else {
                tr[u] = {l, r, 1, 0, 0};
            }
            RE;
        }

        int mid = (l + r) / 2;
        build(u << 1, l, mid, s), build(u << 1 | 1, mid + 1, r, s);
        pushup(u);
    }

    void modify(int u, int l, int r) {
        if (tr[u].l > r || tr[u].r < l) RE;
        if (tr[u].l >= l && tr[u].r <= r) {
            evel(u, 1);
            RE;
        }

        pushdown(u);
        modify(u << 1, l, r);
        modify(u << 1 | 1, l, r);
        pushup(u);
    }

    int query(int u, int l, int r) {
        if (tr[u].l > r || tr[u].r < l) return 0;
        if (tr[u].l >= l && tr[u].r <= r) {
            return tr[u].c1;
        }

        pushdown(u);
        return query(u << 1, l, r) + query(u << 1 | 1, l, r);
    }

} A;

void solve() {
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    s = ' ' + s;

    A.build(1, 1, n, s);

    while (m--) {
        int op, l, r;

        if (op == 1) {
            A.modify(1, l, r);

        } else {
            cout << A.query(1, l, r) << endl;
        }
    }
}
