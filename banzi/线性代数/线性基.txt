//不带时间（删除的）
struct Linear_basis {
    static constexpr int N = 55;
    int num[N];

    bool insert(int x) {
        FORD(i, 0, 49) {
            if (x >> i & 1) {
                if (!num[i]) {
                    num[i] = x;
                    return true;
                }
                x ^= num[i];
            }
        }
        return false;
    }

    int querymin(int x) {
        FORD(i, 0, 49) {
            x = min(x, x ^ num[i]);
        }
        return x;
    }

    int querymax(int x) {
        FORD(i, 0, 49) {
            x = max(x, x ^ num[i]);
        }
        return x;
    }

} A;

//带删除的

struct Linear_basis {
    static constexpr int N = 55;
    int num[N],t[N];

    void insert(int x,int t_) {
        FORD(i, 0, 49) {
            if (x >> i & 1) {
                if (!num[i]) {
                    t[i]=_t;
                    num[i] = x;
                    return;
                }else{
                    if(t[i]<t_){
                        swap(t[i],t_);
                        swap(x,num[i]);
                    }

                    x ^= num[i];
                }
            }
        }
    }

    int querymin(int x,int _t) {
        FORD(i, 0, 49) {
            if(t[i]>=_t)x = min(x, x ^ num[i]);
        }
        return x;
    }

    int querymax(int x,int t_) {
        FORD(i, 0, 49) {
            if(t[i]>=t_)x = max(x, x ^ num[i]);
        }
        return x;
    }

} A;
